# Relax an 8-copy (2x2x2) supercell: close z-layer gaps via deform, NVT settle.
# Default: gap=1.0 A per seam with nz=2 -> deform -2.0 A over the run.
variable timestep equal 0.0005
variable tdamp equal 10*${timestep}
variable force_sq atom "fx^2+fy^2+fz^2"
units metal
newton on
dimension 3
boundary p p p
atom_style atomic
atom_modify map yes
neighbor 2.0 bin
neigh_modify one 3000 page 70000 delay 10 every 10 check yes
thermo 50
read_data ${cfgfile}
mass 1 1.008
timestep ${timestep}
compute total_force_sq all reduce sum v_force_sq
thermo_style custom step time temp pe ke etotal vol density lx ly lz xy xz yz press pxx pyy pzz pxy pxz pyz c_total_force_sq
pair_style mliap unified /work/hdd/bcqo/isaitov/HYDROGEN/M29_elite_LLPT-07-29-2025.model-mliap_lammps.pt 0
pair_coeff * * H
dump 1 all custom 100 dump.relax.atom id type x y z
dump_modify 1 sort id
velocity all create ${temp} 1928374 dist gaussian
velocity all zero linear
velocity all zero angular
fix 1 all nvt temp ${temp} ${temp} ${tdamp}
# Close layer gaps (set deform_dz via -v, e.g. -2.0 for gap=1 / nz=2)
fix 2 all deform 1 z delta 0 ${deform_dz}
run 2000
unfix 1
unfix 2
# Extra settle at fixed box after seams meet
fix 3 all nvt temp ${temp} ${temp} ${tdamp}
run 2000
unfix 3
write_data data.relaxed.txt
write_restart restart.relax.0
